Ejemplo 1: Validar Formulario de Agricultor
python
from utils.validators import (
    validar_texto_requerido,
    validar_identificacion,
    validar_telefono,
    validar_email
)

def validar_formulario_agricultor(datos):
    errores = []
    
    # Nombre
    valido, msg = validar_texto_requerido(datos['nombre'], "Nombre")
    if not valido:
        errores.append(msg)
    
    # Identificación
    valido, msg = validar_identificacion(datos['ci'])
    if not valido:
        errores.append(msg)
    
    # Teléfono (opcional pero si existe, validar)
    if datos.get('telefono'):
        valido, msg = validar_telefono(datos['telefono'])
        if not valido:
            errores.append(msg)
    
    # Email (opcional)
    if datos.get('email'):
        valido, msg = validar_email(datos['email'])
        if not valido:
            errores.append(msg)
    
    return len(errores) == 0, errores

from backend.core.config_manager import ConfigManager
from backend.core.db_installer import DatabaseInstaller

def setup_inicial_completo():
    # 1. Crear configuración
    manager = ConfigManager()
    
    if manager.es_primera_vez():
        print("🔧 Primera ejecución - Configurando sistema...")
        
        # Crear .env
        manager.crear_configuracion(
            server="localhost\\SQLEXPRESS",
            database="Produccion_Agro",
            empresa_nombre="Mi Finca",
            empresa_ruc="987654321"
        )
        
        # 2. Instalar base de datos
        installer = DatabaseInstaller()
        exito, resultado = installer.setup_completo(
            "localhost\\SQLEXPRESS",
            "Produccion_Agro"
        )
        
        if exito:
            print("✅ Sistema configurado")
            print(f"👤 Usuario: {resultado['credenciales']['usuario']}")
            print(f"🔑 Password: {resultado['credenciales']['password']}")
            
            # 3. Marcar setup completado
            manager.marcar_setup_completado()
        else:
            print("❌ Error en setup")
            return False
    
    return True
